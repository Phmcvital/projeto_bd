Table PESSOA {
  id_pessoa int [primary key, increment]
  nome string [not null]
  CPF varchar(11) [unique, not null]
  data_nascimento date
  is_flamengo boolean
  telefone varchar(20)
}

Table PACIENTE {
  id_pessoa int [pk]
  num_convenio varchar
  alergias string
  grupo_sanguineo string(2)
}

Table PROFISSIONAL {
  id_pessoa int [pk]
  CRM varchar
  data_admissao date
  especialidade string
}
Table PRECEPTOR {
  id_profissional int [pk]
  titulacao string
}
Table RESIDENTE {
  id_profissional int [pk]
  ano_residencia varchar(4)
}

Table UNIDADE {
  id_unidade int [pk]
  nome string [not null]
  capacidade_leitos int
}

Table ATENDIMENTO {
  id_atendimento int [pk]
  data_hora date
  duracao_minutos float
  id_paciente int
  id_residente int [not null]
  id_preceptor int [not null]
  id_unidade int
}

Table PROCEDIMENTO {
  id_procedimento        int         [pk, increment]
  nome                   varchar(150) [not null]
  nivel_risco            varchar(5)   [not null, note: 'BAIXO | MEDIO | ALTO']
  tempo_estimado_minutos int
}

Table PROCEDIMENTO_REALIZADO {
  id_atendimento     int     [not null]
  id_procedimento    int     [not null]
  quantidade         int     [not null, default: 1]
  tempo_real_minutos int
  observacao         text
  faturado           boolean [default: false]

  indexes {
    (id_atendimento, id_procedimento) [pk]
  }
}

Table ESCALA_PLANTAO {
  id_escala        int       [pk, increment]
  id_residente     int       [not null]
  id_unidade       int       [not null]
  data_hora_inicio timestamp [not null]
  data_hora_fim    timestamp [not null]
}

Ref: PACIENTE.id_pessoa      - PESSOA.id_pessoa
Ref: PROFISSIONAL.id_pessoa  - PESSOA.id_pessoa

Ref: PRECEPTOR.id_profissional - PROFISSIONAL.id_pessoa
Ref: RESIDENTE.id_profissional - PROFISSIONAL.id_pessoa

Ref: ATENDIMENTO.id_residente > RESIDENTE.id_profissional
Ref: ATENDIMENTO.id_preceptor > PRECEPTOR.id_profissional
Ref: ATENDIMENTO.id_unidade   > UNIDADE.id_unidade

// PROCEDIMENTO_REALIZADO
Ref: PROCEDIMENTO_REALIZADO.id_atendimento  > ATENDIMENTO.id_atendimento
Ref: PROCEDIMENTO_REALIZADO.id_procedimento > PROCEDIMENTO.id_procedimento

// ESCALA_PLANTAO
Ref: ESCALA_PLANTAO.id_residente > RESIDENTE.id_profissional
